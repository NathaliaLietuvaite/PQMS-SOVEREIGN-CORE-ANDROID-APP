class FalsificationLedger:
    """
    Register M / I:
    Self-verifying append-only SHA-256 chain storing full binary state payloads
    (`delta_vector_bytes`), anchored to the hardware Genesis Seal of |L⟩.
    """

    def __init__(self, genesis_seal: str) -> None:
        self._genesis_seal = genesis_seal
        self._genesis_root_hash = hashlib.sha256(genesis_seal.encode("utf-8")).hexdigest()
        self._current_hash = self._genesis_root_hash
        self._entries: List[LedgerEntry] = []
        self._lock = threading.Lock()

    @staticmethod
    def _compute_entry_hash(
        prev_hash: str,
        timestamp_ns: int,
        rcf_direct: float,
        status: str,
        context_tag: str,
        delta_vector_bytes: bytes,
    ) -> str:
        payload = (
            prev_hash.encode("ascii")
            + struct.pack("<Qd", timestamp_ns, rcf_direct)
            + status.encode("utf-8")
            + b"|"
            + context_tag.encode("utf-8")
            + delta_vector_bytes
        )
        return hashlib.sha256(payload).hexdigest()

    def record_transition(
        self, state_vector: np.ndarray, decision: GateDecision, context_tag: str
    ) -> str:
        with self._lock:
            ts = time.time_ns()
            vec_bytes = np.ascontiguousarray(state_vector, dtype=np.float32).tobytes()
            next_hash = self._compute_entry_hash(
                prev_hash=self._current_hash,
                timestamp_ns=ts,
                rcf_direct=decision.rcf_direct,
                status=decision.status.value,
                context_tag=context_tag,
                delta_vector_bytes=vec_bytes,
            )
            entry = LedgerEntry(
                index=len(self._entries),
                prev_hash=self._current_hash,
                entry_hash=next_hash,
                timestamp_ns=ts,
                status=decision.status.value,
                rcf_direct=decision.rcf_direct,
                context_tag=context_tag,
                delta_vector_bytes=vec_bytes,
            )
            self._entries.append(entry)
            self._current_hash = next_hash
            return next_hash

    def verify_chain(self, expected_genesis_seal: Optional[str] = None) -> Dict[str, Any]:
        with self._lock:
            seal = expected_genesis_seal if expected_genesis_seal is not None else self._genesis_seal
            expected_root = hashlib.sha256(seal.encode("utf-8")).hexdigest()
            if expected_root != self._genesis_root_hash:
                return {"verified": False, "entries_checked": 0, "error": "Genesis root mismatch."}

            running_hash = expected_root
            for idx, entry in enumerate(self._entries):
                if entry.index != idx or entry.prev_hash != running_hash:
                    return {"verified": False, "entries_checked": idx, "error": f"Link break at {idx}."}
                recomputed = self._compute_entry_hash(
                    prev_hash=entry.prev_hash,
                    timestamp_ns=entry.timestamp_ns,
                    rcf_direct=entry.rcf_direct,
                    status=entry.status,
                    context_tag=entry.context_tag,
                    delta_vector_bytes=entry.delta_vector_bytes,
                )
                if recomputed != entry.entry_hash:
                    return {"verified": False, "entries_checked": idx, "error": f"Hash mismatch at {idx}."}
                running_hash = recomputed

            return {
                "verified": running_hash == self._current_hash,
                "genesis_root_hash": self._genesis_root_hash,
                "tip_hash": self._current_hash,
                "entries_checked": len(self._entries),
            }
